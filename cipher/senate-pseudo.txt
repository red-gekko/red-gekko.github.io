ALPHABETCAPS = (ABCDEFGHIJKLMNOPQRSTUVWXYZ)
NUMERIC = (0123456789)
PRINT "WELCOME"
FUNCTION "JULIUS"
	KEY = INPUT(ENTER KEY)[NUMBER ONLY]
	MESSAGE = INPUT(ENTER MESSAGE)
	NEWMESSAGE = ""
	TAKE CHARACTERS FROM MESSAGE
		IF CHARACTERS ARE IN ALPHABET
			FIND POSITION
			NEWPOSITION = POSITION + KEY, LOOP BACK ON 26
			NEWCHARACTER = NEWPOSTION IN ALPHABET
			ADD NEWCHARACTER TO NEWMESSAGE
		ELSE IF CHARACTERS ARE IN ALPHABETCAPS
			FIND POSITION
			NEWPOSITION = POSITION + KEY, LOOP BACK ON 26
			NEWCHARACTER = NEWPOSTION IN ALPHABETCAPS
			ADD NEWCHARACTER TO NEWMESSAGE
		ELSE IF CHARACTERS ARE IN NUMERIC
			FIND POSITION
			NEWPOSITION = POSITION + KEY, LOOP BACK ON 10
			NEWCHARACTER = NEWPOSTION IN NUMERIC
			ADD NEWCHARACTER TO NEWMESSAGE
		ELSE
			ADD CHARACTER TO NEWMESSAGE
	PRINT NEWMESSAGE
	EXECUTE SENATE

FUNCTION SENATE
	CHOICE = INPUT (JULIUS TO ECRYPT, BRUTUS TO DECRYPT, EXIT TO EXIT)
	IF CHOICE IS JULIUS
		EXECUTE JULIUS
	ELSE IF CHOICE IS BRUTUS
		EXECUTE BRUTUS
	ELSE IF CHOICE IS EXIT
		EXIT
	ELSE
		PRINT "INVALID COMMMAND"
		EXECUTE SENATE
EXECUTE SENATE
